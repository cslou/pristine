-- Spec-005 §12 indexes (ix_*) --

ix_conversations_project_started  on  conversations
  CREATE INDEX ix_conversations_project_started
  ON conversations(project_id, created_at DESC)

ix_messages_conv_sort  on  messages
  CREATE INDEX ix_messages_conv_sort
  ON messages(conversation_id, sort_order)

ix_messages_parent  on  messages
  CREATE INDEX ix_messages_parent
  ON messages(parent_message_id)

ix_messages_timestamp_nonchunk  on  messages
  CREATE INDEX ix_messages_timestamp_nonchunk
  ON messages(timestamp DESC) WHERE parent_message_id IS NULL

ix_summaries_project_time  on  summaries
  CREATE INDEX ix_summaries_project_time
  ON summaries(project_id, timestamp DESC)

ix_window_messages_message_id  on  window_messages
  CREATE INDEX ix_window_messages_message_id
  ON window_messages(message_id)
